# Plan 2.3: AI Response JSON Parsing, Streaming, and Typewriter Effect

## Objective

Extend the AI layer with streaming support and structured response parsing. Add a `StreamProvider` interface alongside the existing `Provider`. Implement a typewriter Bubbletea component that renders text character-by-character. Create a response parser that extracts structured JSON (narrative, choices, state_changes, mood) from AI output. After this plan, the system can stream AI text into the TUI with a typewriter effect and parse the structured gameplay JSON format.

## must_haves

- `StreamProvider` interface defined with a `Stream` method returning a channel
- At least the OpenAI-compatible provider (LiteLLM) supports streaming
- Response parser extracts `NarrativeResponse` from AI JSON output
- Typewriter component renders text incrementally at configurable speed
- Typewriter works as a standalone Bubbletea model (reusable in narrative view)
- Non-streaming providers fall back to full-text delivery (simulated streaming)
- Router tries streaming first, falls back to non-streaming Complete

```go
// StreamChunk is a piece of streamed AI response.
type StreamChunk struct {
    Content string // incremental text
    Done    bool   // true on final chunk
    Error   error  // non-nil if stream errored
}

// StreamProvider extends Provider with streaming capability.
type StreamProvider interface {
    Provider
    // Stream sends a request and returns a channel of incremental chunks.
    // The channel is closed when streaming completes.
    // The caller must read all chunks (or cancel the context).
    Stream(ctx context.Context, req Request) (<-chan StreamChunk, error)
}
```

```go
// Stream implements ai.StreamProvider for SSE streaming.
func (p *OpenAICompatProvider) Stream(ctx context.Context, req ai.Request) (<-chan ai.StreamChunk, error) {
    // Build the request body same as Complete but with "stream": true
    body := map[string]interface{}{
        "model":       p.model(req),
        "messages":    req.Messages,
        "temperature": req.Temperature,
        "max_tokens":  req.MaxTokens,
        "stream":      true,
    }

    jsonBody, err := json.Marshal(body)
    if err != nil {
        return nil, fmt.Errorf("marshaling request: %w", err)
    }

    httpReq, err := http.NewRequestWithContext(ctx, "POST", p.baseURL+"/chat/completions", bytes.NewReader(jsonBody))
    if err != nil {
        return nil, fmt.Errorf("creating request: %w", err)
    }
    httpReq.Header.Set("Content-Type", "application/json")
    if p.apiKey != "" {
        httpReq.Header.Set("Authorization", "Bearer "+p.apiKey)
    }

    resp, err := p.client.Do(httpReq)
    if err != nil {
        return nil, fmt.Errorf("sending request: %w", err)
    }

    if resp.StatusCode != http.StatusOK {
        resp.Body.Close()
        return nil, fmt.Errorf("API returned status %d", resp.StatusCode)
    }

    ch := make(chan ai.StreamChunk, 32)

    go func() {
        defer close(ch)
        defer resp.Body.Close()

        scanner := bufio.NewScanner(resp.Body)
        for scanner.Scan() {
            line := scanner.Text()
            if !strings.HasPrefix(line, "data: ") {
                continue
            }
            data := strings.TrimPrefix(line, "data: ")
            if data == "[DONE]" {
                ch <- ai.StreamChunk{Done: true}
                return
            }

            var chunk struct {
                Choices []struct {
                    Delta struct {
                        Content string `json:"content"`
                    } `json:"delta"`
                } `json:"choices"`
                Model string `json:"model"`
            }
            if err := json.Unmarshal([]byte(data), &chunk); err != nil {
                continue // skip malformed chunks
            }
            if len(chunk.Choices) > 0 && chunk.Choices[0].Delta.Content != "" {
                ch <- ai.StreamChunk{Content: chunk.Choices[0].Delta.Content}
            }
        }
        if err := scanner.Err(); err != nil {
            ch <- ai.StreamChunk{Error: err}
        }
        ch <- ai.StreamChunk{Done: true}
    }()

    return ch, nil
}
```

Add `bufio` to imports. This uses standard SSE parsing for OpenAI-compatible streaming endpoints (works with LiteLLM and OpenRouter).
</action>

```go
// Stream tries each provider in order, preferring StreamProvider.
// If no provider supports streaming, it falls back to Complete and simulates a stream.
func (r *Router) Stream(ctx context.Context, req Request) (<-chan StreamChunk, string, error) {
    var errors []string

    // First pass: try StreamProvider implementations
    for _, p := range r.providers {
        if sp, ok := p.(StreamProvider); ok {
            ch, err := sp.Stream(ctx, req)
            if err != nil {
                errors = append(errors, fmt.Sprintf("%s(stream): %s", p.Name(), err))
                continue
            }
            return ch, p.Name(), nil
        }
    }

    // Second pass: fall back to Complete and simulate streaming
    for _, p := range r.providers {
        resp, err := p.Complete(ctx, req)
        if err != nil {
            errors = append(errors, fmt.Sprintf("%s: %s", p.Name(), err))
            continue
        }
        // Simulate streaming by sending the full response as one chunk
        ch := make(chan StreamChunk, 2)
        go func() {
            ch <- StreamChunk{Content: resp.Content}
            ch <- StreamChunk{Done: true}
            close(ch)
        }()
        return ch, p.Name(), nil
    }

    return nil, "", fmt.Errorf("all AI providers failed:\n  %s", strings.Join(errors, "\n  "))
}
```

The third return value is the provider name so the status bar can show it.
</action>

```go
package ai

import (
    "encoding/json"
    "regexp"
)

// jsonBlockRe matches ```json ... ``` blocks in AI output.
var jsonBlockRe = regexp.MustCompile("(?s)```json\\s*\\n(.*?)\\n```")

// ParseNarrativeJSON extracts a NarrativeResponse-shaped JSON from AI text.
// Returns the raw JSON string and any parse error.
// If no JSON block is found, returns empty string and nil error (pure narrative).
func ParseNarrativeJSON(text string) (string, error) {
    matches := jsonBlockRe.FindStringSubmatch(text)
    if len(matches) < 2 {
        return "", nil
    }
    // Validate it's valid JSON
    var raw json.RawMessage
    if err := json.Unmarshal([]byte(matches[1]), &raw); err != nil {
        return "", err
    }
    return matches[1], nil
}

// ExtractNarrative returns just the narrative text, stripping JSON blocks.
// This is useful when the AI response has both prose and a JSON block.
func ExtractNarrative(text string) string {
    return jsonBlockRe.ReplaceAllString(text, "")
}
```

```go
package components

import (
    "time"

    tea "github.com/charmbracelet/bubbletea"
)

// TypewriterDoneMsg signals the typewriter has finished rendering all text.
type TypewriterDoneMsg struct{}

// typewriterTickMsg is the internal tick for advancing characters.
type typewriterTickMsg struct{}

// TypewriterModel renders text character-by-character.
type TypewriterModel struct {
    fullText    string        // the complete text to render
    displayed   int           // how many characters are currently shown
    speed       time.Duration // delay between characters
    active      bool          // whether the typewriter is running
    done        bool          // all text displayed
}

// NewTypewriter creates a typewriter component.
// speed is characters per second (e.g., 80 = 80 chars/sec).
func NewTypewriter(speed int) TypewriterModel {
    if speed <= 0 {
        speed = 80
    }
    return TypewriterModel{
        speed: time.Second / time.Duration(speed),
    }
}

// SetText sets new text and starts the typewriter effect.
// Returns a Cmd to start ticking.
func (t *TypewriterModel) SetText(text string) tea.Cmd {
    t.fullText = text
    t.displayed = 0
    t.active = true
    t.done = false
    return t.tick()
}

// AppendText adds more text (for streaming) and continues the effect.
func (t *TypewriterModel) AppendText(text string) tea.Cmd {
    t.fullText += text
    if !t.active && !t.done {
        t.active = true
        return t.tick()
    }
    // If already active, ticks will naturally catch up
    return nil
}

// Skip immediately displays all text.
func (t *TypewriterModel) Skip() {
    t.displayed = len([]rune(t.fullText))
    t.active = false
    t.done = true
}

// IsActive returns true if the typewriter is still animating.
func (t TypewriterModel) IsActive() bool {
    return t.active
}

// IsDone returns true if all text has been displayed.
func (t TypewriterModel) IsDone() bool {
    return t.done
}

// View returns the currently visible text.
func (t TypewriterModel) View() string {
    runes := []rune(t.fullText)
    if t.displayed >= len(runes) {
        return t.fullText
    }
    return string(runes[:t.displayed])
}

// Update handles tick messages.
func (t TypewriterModel) Update(msg tea.Msg) (TypewriterModel, tea.Cmd) {
    switch msg.(type) {
    case typewriterTickMsg:
        if !t.active {
            return t, nil
        }
        runes := []rune(t.fullText)
        t.displayed++
        if t.displayed >= len(runes) {
            t.displayed = len(runes)
            t.active = false
            t.done = true
            return t, func() tea.Msg { return TypewriterDoneMsg{} }
        }
        return t, t.tick()
    }
    return t, nil
}

func (t TypewriterModel) tick() tea.Cmd {
    return tea.Tick(t.speed, func(time.Time) tea.Msg {
        return typewriterTickMsg{}
    })
}
```
